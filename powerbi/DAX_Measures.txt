-- Create these measures in the Sales_Order_Items table (or a dedicated Measures table)
Revenue = SUM(Sales_Order_Items[Revenue])
Gross Sales = SUM(Sales_Order_Items[Gross_Sales])
Discount Amount = SUM(Sales_Order_Items[Discount_Amount])
Cost = SUM(Sales_Order_Items[Cost])
Profit = SUM(Sales_Order_Items[Profit])
Profit Margin % = DIVIDE([Profit],[Revenue])
Discount Rate % = DIVIDE([Discount Amount],[Gross Sales])
Units Sold = SUM(Sales_Order_Items[Quantity])
Order Count = DISTINCTCOUNT(Sales_Orders[Order_ID])
Average Order Revenue = DIVIDE([Revenue],[Order Count])
Returned Refunds = SUM(Returns[Refund_Amount])
Return Count = COUNTROWS(Returns)
Return Rate % = DIVIDE([Return Count], DISTINCTCOUNT(Sales_Order_Items[Order_Item_ID]))
Low Margin Lines = CALCULATE(COUNTROWS(Sales_Order_Items), FILTER(Sales_Order_Items, DIVIDE(Sales_Order_Items[Profit],Sales_Order_Items[Revenue]) < 0.10))
Target Revenue = SUM(Sales_Targets[Target_Revenue])
Target Variance = [Revenue] - [Target Revenue]
Target Achievement % = DIVIDE([Revenue],[Target Revenue])
